        def run_simple(src_folder):

    resized_frames, frame_ids, K, k1, k2, shot_params = get_data(src_folder)

    ref_frame = resized_frames[0]
    R, ref_t = np.array(shot_params[frame_ids[0]]['R']).reshape(3, 1), np.array(shot_params[frame_ids[0]]['T']).reshape(3, 1)
    ref_R, _ = cv.Rodrigues(R)

    # Initialize panorama
    panorama = ref_frame

    for i in range(1, len(resized_frames)):
        # Calculate relative R and t
        
        Rs, ts = np.array(shot_params[frame_ids[i]]['R']).reshape(3, 1), np.array(shot_params[frame_ids[i]]['T']).reshape(3, 1)
        Rs, _ = cv.Rodrigues(R)
        
        R_relative = Rs @ ref_R.T
        t_relative = ts - R_relative @ ref_t

        # Calculate homography
        H = K @ (R_relative + t_relative @ np.array([[0, 0, 1]])) @ np.linalg.inv(K)

        print('**************************')
        print(' ---- Homography-Simple ---- ')
        print(H)
        print('**************************')

        # Warp current frame
        warped_frame = cv2.warpPerspective(resized_frames[i], H, (panorama.shape[1]*2, panorama.shape[0]*2))        

        # Blend warped frame with panorama (simple overlay for demonstration)
        
        # fig = plt.figure(figsize = (7, 7))
        # plt.imshow(warped_frame)
        # plt.show()
        
        panorama = warped_frame   

    # Crop black borders (simplified)
    gray = cv2.cvtColor(panorama, cv2.COLOR_BGR2GRAY)
    _, thresh = cv2.threshold(gray, 1, 255, cv2.THRESH_BINARY)
    contours, _ = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    print(contours)
    x, y, w, h = cv2.boundingRect(contours[0])
    panorama = panorama[y:y+h, x:x+w]

    return panorama
